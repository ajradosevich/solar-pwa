async function fetchSolarData() { const secretKey = getAppSecret(false); if (!secretKey) { document.getElementById('status-dot').style.backgroundColor = '#ef4444'; return; } const overlay = document.getElementById('loading-overlay'); overlay.classList.add('visible'); try { const url = `${APPS_SCRIPT_URL}?mode=${currentMode}&offset=${currentOffset}&key=${encodeURIComponent(secretKey)}&_t=${Date.now()}`; const response = await fetch(url); const data = await response.json(); if (data.status === 'unauthorized' || data.error === 'Unauthorized') { document.getElementById('status-dot').style.backgroundColor = '#ef4444'; alert('Unauthorized: Incorrect App Secret key.'); const retryKey = getAppSecret(true); if (retryKey) fetchSolarData(); return; } if (data.status === 'error' || data.error) { throw new Error(data.message || data.error || 'Backend error'); } if (data.periodLabel) { document.getElementById('period-label').textContent = data.periodLabel; } updateKPIs(data); renderChart(data); document.getElementById('status-dot').style.backgroundColor = '#10b981'; } catch (err) { console.error('Error fetching solar data:', err); alert('Error: ' + err.message); document.getElementById('status-dot').style.backgroundColor = '#ef4444'; } finally { overlay.classList.remove('visible'); } } function updateKPIs(data) { // Support both V5 (data.summary) and older flat Code.gs payloads const s = data.summary || data.totals || data || {}; const latest = data.latest || data.current || null; const winSolar = Number(s.windowSolarKwh ?? s.dailySolarKwh ?? s.solarKwh ?? 0); const peakKw = Number(s.peakSolarKw ?? s.peakKw ?? 0); const winHome = Number(s.windowHomeKwh ?? s.dailyHomeKwh ?? s.homeKwh ?? 0); const netKwh = Number(s.windowNetKwh ?? s.dailyNetKwh ?? (winHome - winSolar)); document.getElementById('kpi-solar').textContent = `${winSolar.toFixed(2)} kWh`; document.getElementById('kpi-peak').textContent = `Peak: ${peakKw.toFixed(2)} kW`; document.getElementById('kpi-home').textContent = `${winHome.toFixed(2)} kWh`; const netEl = document.getElementById('kpi-net'); const netDirEl = document.getElementById('kpi-net-dir'); netEl.textContent = `${Math.abs(netKwh).toFixed(2)} kWh`; if (netKwh <= 0) { netEl.className = 'kpi-value val-export'; netDirEl.textContent = 'Net Exported to Grid'; } else { netEl.className = 'kpi-value val-import'; netDirEl.textContent = 'Net Imported from Grid'; } if (latest) { const liveSolar = Number(latest.solarKw ?? latest.solarProduction ?? latest.solar ?? 0); const liveHome = Number(latest.homeKw ?? latest.homeUsage ?? latest.home ?? 0); const gridV12 = Number(latest.gridV12 ?? 0); const maxTempC = Number(latest.maxTempC ?? 0); const prodLtea = Number(latest.prodLteaKwh ?? 0); document.getElementById('kpi-live-solar').textContent = `${liveSolar.toFixed(2)} kW`; document.getElementById('kpi-live-home').textContent = `Live: ${liveHome.toFixed(2)} kW`; if (latest.timestamp) { const tStr = new Date(latest.timestamp).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }); document.getElementById('kpi-last-seen').textContent = `Last log: ${tStr}`; } const invEl = document.getElementById('diag-inv'); if (latest.invTotal != null && Number(latest.invTotal) > 0) { invEl.textContent = `${latest.invOnline ?? 0} / ${latest.invTotal}`; invEl.style.color = (latest.invOnline === latest.invTotal) ? '#10b981' : '#f59e0b'; document.getElementById('diag-spread').textContent = `Spread: ${Number(latest.spreadW ?? 0).toFixed(0)} W`; } if (gridV12 > 0) { document.getElementById('diag-v12').textContent = `${gridV12.toFixed(1)} V`; } if (maxTempC > 0) { const tempF = (maxTempC * 9 / 5) + 32; document.getElementById('diag-temp').textContent = `${maxTempC.toFixed(1)} °C`; document.getElementById('diag-temp-f').textContent = `${tempF.toFixed(1)} °F`; } if (prodLtea > 0) { const mwh = prodLtea / 1000.0; document.getElementById('ft-ltea').textContent = `${mwh.toFixed(2)} MWh`; } } const todaySolar = Number(s.todaySolarKwh ?? s.dailySolarKwh ?? winSolar); const monthSolar = Number(s.monthSolarKwh ?? s.monthlySolarKwh ?? 0); document.getElementById('ft-today').textContent = `${todaySolar.toFixed(2)} kWh`; document.getElementById('ft-month').textContent = `${monthSolar.toFixed(2)} kWh`; const seriesArr = data.series || data.history || data.data || []; document.getElementById('chart-points').textContent = `${seriesArr.length} intervals`; } function renderChart(data) { const ctx = document.getElementById('solarChart').getContext('2d'); const series = data.series || data.history || data.data || []; const labels = series.map(pt => pt.label || pt.time || ''); const solarVals = series.map(pt => Number(pt.solar ?? pt.solarProduction ?? 0)); const homeVals = series.map(pt => Number(pt.home ?? pt.homeUsage ?? 0)); const isDayMode = (data.mode || currentMode) === 'day'; const chartType = isDayMode ? 'line' : 'bar'; const unit = data.unit || (isDayMode ? 'kW' : 'kWh'); if (solarChartInstance) { solarChartInstance.destroy(); } solarChartInstance = new Chart(ctx, { type: chartType, data: { labels: labels, datasets: [ { label: `Solar (${unit})`, data: solarVals, borderColor: '#f59e0b', backgroundColor: isDayMode ? 'rgba(245, 158, 11, 0.18)' : 'rgba(245, 158, 11, 0.85)', borderWidth: isDayMode ? 2 : 0, borderRadius: isDayMode ? 0 : 4, fill: isDayMode, tension: 0.25, pointRadius: 0, pointHitRadius: 10 }, { label: `Home (${unit})`, data: homeVals, borderColor: '#38bdf8', backgroundColor: isDayMode ? 'rgba(56, 189, 248, 0.10)' : 'rgba(56, 189, 248, 0.75)', borderWidth: isDayMode ? 1.8 : 0, borderRadius: isDayMode ? 0 : 4, fill: isDayMode, tension: 0.25, pointRadius: 0, pointHitRadius: 10 } ] }, options: { responsive: true, maintainAspectRatio: false, interaction: { mode: 'index', intersect: false }, plugins: { legend: { labels: { color: '#e8eaed', boxWidth: 12, font: { size: 11 } } }, tooltip: { backgroundColor: '#111318', titleColor: '#e8eaed', bodyColor: '#e8eaed', borderColor: '#2b303c', borderWidth: 1, callbacks: { label: function(context) { const val = Number(context.parsed?.y ?? 0); return `${context.dataset.label}: ${val.toFixed(2)} ${unit}`; } } } }, scales: { x: { grid: { color: 'rgba(255, 255, 255, 0.05)' }, ticks: { color: '#9aa0a6', maxRotation: 0, autoSkip: true, maxTicksLimit: isDayMode ? 8 : 10, font: { size: 10 } } }, y: { beginAtZero: true, grid: { color: 'rgba(255, 255, 255, 0.06)' }, ticks: { color: '#9aa0a6', font: { size: 10 }, callback: value => `${value} ${unit}` } } } } }); }
